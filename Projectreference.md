The canonical, hardened rules used by both RAMP Admin and RAMP are in

      rules_version = '2';
      service cloud.firestore {
      match /databases/{database}/documents {
      function signedIn() {
      return request.auth != null;
      }

    function hasProfile(uid) {
      return exists(/databases/$(database)/documents/users/$(uid));
    }

    function profile(uid) {
      return get(/databases/$(database)/documents/users/$(uid)).data;
    }

    function hasActiveLandlordRecord(uid) {
      return exists(/databases/$(database)/documents/landlords/$(uid)) &&
        get(/databases/$(database)/documents/landlords/$(uid))
          .data.status == 'active';
    }

      function isSuperAdmin() {
      return signedIn() && (
      request.auth.token.get('role', '') == 'super_admin' ||
      (hasProfile(request.auth.uid) && profile(request.auth.uid).get('role', '') == 'super_admin') ||
      (exists(/databases/$(database)/documents/adminProfiles/$(request.auth.uid)) &&
      get(/databases/$(database)/documents/adminProfiles/$(request.auth.uid)).data.get('status', 'active') == 'active')
      );
      }


    function isActiveLandlord() {
      return signedIn() &&
        hasProfile(request.auth.uid) &&
        profile(request.auth.uid).get('role', '') == 'landlord' &&
        profile(request.auth.uid).get('isApproved', true) == true &&
        hasActiveLandlordRecord(request.auth.uid);
    }

    function isTenant() {
      return signedIn() &&
        hasProfile(request.auth.uid) &&
        profile(request.auth.uid).get('role', '') == 'tenant' &&
        profile(request.auth.uid).get('tenantId', '') is string &&
        profile(request.auth.uid).get('tenantId', '') != '';
    }

    function linkedTenantId() {
      return profile(request.auth.uid).get('tenantId', '');
    }

    function ownsNewRecord() {
      return request.resource.data.get('landlordId', '') == request.auth.uid;
      }
      
      function ownsExistingRecord() {
      // Use empty string fallback instead of request.auth.uid fallback
      return resource.data.get('landlordId', '') == request.auth.uid &&
      request.resource.data.get('landlordId', '') == request.auth.uid;
      }

    function ownsOrMigratesLegacyRecord() {
      return (resource.data.get('landlordId', request.auth.uid) == request.auth.uid ||
          resource.data.get('landlordId', null) == null) &&
        request.resource.data.get('landlordId', '') == request.auth.uid;
    }

    function isOperationalCollection(collection) {
      return collection in [
        'units', 'tenants', 'payments', 'maintenanceTickets',
        'announcements', 'notifications', 'expenses', 'tenantDocuments',
        'settings', 'utilityRateLogs', 'events'
      ];
    }

    function canTenantRead(collection, documentId) {
      return isTenant() && (
        (collection == 'tenants' && documentId == linkedTenantId()) ||
        (collection in [
          'units', 'payments', 'maintenanceTickets', 'tenantDocuments'
        ] && resource.data.get('tenantId', '') == linkedTenantId()) ||
        collection in ['announcements', 'settings']
      );
    }

    function ownsLinkedUnit(unitId) {
      return isTenant() &&
        unitId is string &&
        exists(/databases/$(database)/documents/tenants/$(linkedTenantId())) &&
        get(/databases/$(database)/documents/tenants/$(linkedTenantId()))
          .data.get('unitId', '') == unitId &&
        exists(/databases/$(database)/documents/units/$(unitId)) &&
        get(/databases/$(database)/documents/units/$(unitId))
          .data.get('tenantId', '') == linkedTenantId();
    }

    match /users/{uid} {
      allow read: if isSuperAdmin() ||
        (signedIn() && request.auth.uid == uid);
      allow create: if isSuperAdmin() ||
        (signedIn() &&
         request.auth.uid == uid &&
           hasActiveLandlordRecord(uid) &&
         request.resource.data.role == 'landlord' &&
         request.resource.data.get('landlordId', '') == uid &&
         request.resource.data.get('isApproved', true) == true &&
         request.resource.data.get('tenantId', null) == null);
      allow update: if isSuperAdmin() ||
        (signedIn() &&
          request.auth.uid == uid &&
          request.resource.data.role == resource.data.role &&
          request.resource.data.get('isApproved', true) ==
            resource.data.get('isApproved', true) &&
          request.resource.data.get('landlordId', null) ==
            resource.data.get('landlordId', null) &&
          request.resource.data.get('tenantId', null) ==
            resource.data.get('tenantId', null));
      allow delete: if isSuperAdmin();
    }

    match /landlords/{uid} {
      allow read: if isSuperAdmin() || (signedIn() && request.auth.uid == uid);
      
      // Allow users to request landlord status, but enforce a 'pending' state
      allow create: if signedIn() &&
      request.auth.uid == uid &&
      request.resource.data.get('status', 'pending') == 'pending';
      
      // Only super admins can approve and change status to 'active'
      allow update, delete: if isSuperAdmin();
      }


    match /adminProfiles/{uid} {
      allow read: if isSuperAdmin() &&
        (request.auth.uid == uid || request.auth.token.get('role', '') in
          ['super_admin', 'superAdmin', 'superadmin']);
      allow write: if false;
    }

    match /adminAuditLogs/{logId} {
      allow read: if isSuperAdmin();
      allow write: if false;
    }

    match /audit_logs/{logId} {
      allow read: if isSuperAdmin() || isActiveLandlord();
      allow create: if isActiveLandlord() && ownsNewRecord();
      allow update: if isActiveLandlord() &&
        request.resource.data == resource.data;
      allow delete: if false;
    }

    match /auditLogs/{logId} {
      allow read, write: if isSuperAdmin() || isActiveLandlord();
    }

    match /payments/{paymentId} {
      allow create: if isSuperAdmin() ||
        (isActiveLandlord() && ownsNewRecord()) ||
        (isTenant() &&
          request.resource.data.get('tenantId', '') == linkedTenantId() &&
          ownsLinkedUnit(request.resource.data.get('unitId', '')) &&
          request.resource.data.get('status', '') == 'Pending');
    }

    match /maintenanceTickets/{ticketId} {
      allow create: if isSuperAdmin() ||
        (isActiveLandlord() && ownsNewRecord()) ||
        (isTenant() &&
          request.resource.data.get('tenantId', '') == linkedTenantId() &&
          ownsLinkedUnit(request.resource.data.get('unitId', '')) &&
          request.resource.data.get('status', '') == 'Pending' &&
          request.resource.data.get('rating', 0) == 0 &&
          request.resource.data.get('estimatedCost', 0) == 0 &&
          request.resource.data.get('actualCost', 0) == 0 &&
          request.resource.data.get('vendorCost', 0) == 0 &&
          request.resource.data.get('paymentRequired', false) == false);
      allow update: if isTenant() &&
        resource.data.get('tenantId', '') == linkedTenantId() &&
        resource.data.get('status', '') in ['Completed', 'Resolved', 'Closed'] &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['rating', 'ratingFeedback']) &&
        request.resource.data.get('rating', 0) is int &&
        request.resource.data.get('rating', 0) >= 1 &&
        request.resource.data.get('rating', 0) <= 5;
    }

    match /{collection}/{documentId} {
      allow read: if isOperationalCollection(collection) &&
        (isSuperAdmin() || isActiveLandlord() ||
          canTenantRead(collection, documentId));
      allow create: if isOperationalCollection(collection) &&
        (isSuperAdmin() ||
          (isActiveLandlord() && ownsNewRecord()));
      allow update: if isOperationalCollection(collection) &&
        (isSuperAdmin() ||
          (isActiveLandlord() &&
            (ownsExistingRecord() || ownsOrMigratesLegacyRecord())));
      allow delete: if isOperationalCollection(collection) &&
        (isSuperAdmin() ||
          (isActiveLandlord() &&
            (resource.data.get('landlordId', null) == request.auth.uid ||
              !resource.data.keys().hasAny(['landlordId']))));

      match /liveLocation/{locationId} {
        allow read, write: if isSuperAdmin() || isActiveLandlord();
      }
    }
}
